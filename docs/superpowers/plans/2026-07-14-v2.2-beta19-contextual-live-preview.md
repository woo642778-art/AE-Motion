# AE motion v2.2 Beta 19 Contextual Live Preview Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `AE-Motion-v2.2-beta19-unsigned.ipa` with BCC Edge Glow thumbnail resolution, stricter effect visual-safety gating, editing controls placed beside their host features, immediate preview updates, and a reduced Extensions & Scripts surface.

**Architecture:** Keep animation and velocity engines in `AEMotionExtensionsCore`, but replace the standalone Animation Core entry with fail-closed contextual UIKit adapters in `AEMotionExtensionsHost`. A preview transaction snapshots recognized host controls, applies user changes through existing `UIControl` value-change paths at display-refresh cadence, invalidates the active preview, and restores the snapshot on cancellation or selection change. Packaging starts from the v2.1 Beta 17 IPA so Beta 18's forced Other assignments are not inherited, applies the v2.2 repair and qualification pipeline, and allows Other membership only when exact descriptor hashes have matching qualification evidence.

**Tech Stack:** Swift 6, UIKit, Objective-C runtime, XCTest, Python 3 standard library, POSIX shell, GitHub Actions on `macos-15`, generic arm64 iOS Release framework build.

## Global Constraints

- Minimum platform remains iOS 15 and macOS 13 as declared in `Package.swift`.
- Display name must remain exactly `AE motion`.
- Bundle identifier must remain exactly `com.alightcreative.motion`.
- The Alight Motion main executable must remain byte-identical to the v2.1 Beta 17 base IPA.
- Do not read or write the host app's private project or timeline database.
- Do not use an export-render-reimport workflow for contextual editing.
- Host integration is allowlisted and fail-closed; unsupported controller or control contracts expose no editing button.
- Keep effect IDs unchanged when repairing an existing effect.
- BCC Edge Glow must remain `com.alightcreative.effects.bccedgeglow`.
- Other membership is evidence-based; there is no minimum or target effect count.
- Effects lacking exact-hash device qualification may remain in their original category, but may not be promoted to Other.
- Effects with a high-confidence runtime, black-bar, blank-output, alpha-loss, or unsafe-coordinate finding are quarantined before signing.
- The output IPA is unsigned and must contain no `_CodeSignature`, `CodeResources`, or `embedded.mobileprovision` material.
- Release notes contain only brief English bullets.

```swift
import Foundation

public enum ContextualEditingContext: String, Codable, CaseIterable, Sendable {
    case transform, graph, speed, effect
}

public struct ContextualToolDescriptor: Identifiable, Equatable, Codable, Sendable {
    public let id: String
    public let title: String
    public let context: ContextualEditingContext
    public let systemImage: String

    public init(id: String, title: String, context: ContextualEditingContext, systemImage: String) {
        self.id = id
        self.title = title
        self.context = context
        self.systemImage = systemImage
    }
}

public enum PreviewTransactionState: String, Codable, Equatable, Sendable {
    case idle, active, committed, cancelled
}

public enum PreviewTransactionError: Error, Equatable {
    case alreadyStarted
    case notActive
}

public final class PreviewTransaction<Value: Equatable>: @unchecked Sendable {
    public private(set) var state: PreviewTransactionState = .idle
    public private(set) var currentValue: Value
    public let initialValue: Value
    private let apply: @Sendable (Value) -> Void

    public init(initialValue: Value, apply: @escaping @Sendable (Value) -> Void) {
        self.initialValue = initialValue
        self.currentValue = initialValue
        self.apply = apply
    }

    public func begin() throws {
        guard state == .idle else { throw PreviewTransactionError.alreadyStarted }
        state = .active
    }

    public func update(_ value: Value) throws {
        guard state == .active else { throw PreviewTransactionError.notActive }
        currentValue = value
        apply(value)
    }

    public func commit() throws {
        guard state == .active else { throw PreviewTransactionError.notActive }
        state = .committed
    }

    public func cancel() throws {
        guard state == .active else { throw PreviewTransactionError.notActive }
        currentValue = initialValue
        apply(initialValue)
        state = .cancelled
    }

    public func selectionDidChange() throws {
        if state == .active { try cancel() }
    }
}
```

import Foundation

public actor PreviewUpdateCoalescer<Value: Sendable> {
    private var pending: Value?

    public init() {}

    public func enqueue(_ value: Value) {
        pending = value
    }

    public func flush(_ deliver: @Sendable (Value) -> Void) {
        guard let value = pending else { return }
        pending = nil
        deliver(value)
    }

    public func finish(_ deliver: @Sendable (Value) -> Void) {
        flush(deliver)
    }
}
```

public struct DeviceQualificationRecord: Equatable, Codable, Sendable {
    public var effectID: String
    public var descriptorSHA256: String
    public var hostVersion: String
    public var deviceClass: String
    public var stage: EffectQualificationStage
    public var disposition: EffectQualificationDisposition
    public var metrics: EffectVisualMetrics
}

public enum EffectQualificationPolicy {
    public static func isEligibleForOther(
        effectID: String,
        descriptorSHA256: String,
        records: [DeviceQualificationRecord]
    ) -> Bool {
        records.contains {
            $0.effectID.caseInsensitiveCompare(effectID) == .orderedSame
                && $0.descriptorSHA256.lowercased() == descriptorSHA256.lowercased()
                && $0.stage == .device
                && $0.disposition == .passed
        }
    }
}

#if canImport(UIKit)
import UIKit
import AEMotionExtensionsCore

@MainActor
struct HostEditingContract {
    let context: ContextualEditingContext
    let requiredTokens: [String]
    let valueControls: [UISlider]
    let previewView: UIView
    let selectionIdentity: ObjectIdentifier

    var isSupported: Bool {
        !requiredTokens.isEmpty && !valueControls.isEmpty && previewView.window != nil
    }
}

@MainActor
protocol HostControlMutationAdapter: AnyObject {
    var context: ContextualEditingContext { get }
    var selectionIdentity: ObjectIdentifier { get }
    func snapshot() -> [Float]
    func apply(_ values: [Float]) -> Bool
    func restore(_ values: [Float])
    func invalidatePreview()
}

@MainActor
final class SliderControlMutationAdapter: HostControlMutationAdapter {
    let context: ContextualEditingContext
    let selectionIdentity: ObjectIdentifier
    private let controls: [UISlider]
    private weak var previewView: UIView?

    init(contract: HostEditingContract) {
        self.context = contract.context
        self.selectionIdentity = contract.selectionIdentity
        self.controls = contract.valueControls
        self.previewView = contract.previewView
    }

    func snapshot() -> [Float] { controls.map(\.value) }

    func apply(_ values: [Float]) -> Bool {
        guard values.count == controls.count else { return false }
        for (control, value) in zip(controls, values) {
            control.setValue(min(max(value, control.minimumValue), control.maximumValue), animated: false)
            control.sendActions(for: .valueChanged)
        }
        invalidatePreview()
        return true
    }

    func restore(_ values: [Float]) {
        _ = apply(values)
    }

    func invalidatePreview() {
        previewView?.setNeedsLayout()
        previewView?.layoutIfNeeded()
        previewView?.setNeedsDisplay()
    }
}
#endif
```

- [ ] **Step 4: Implement live preview coordination**

```swift
#if canImport(UIKit)
import UIKit
import AEMotionExtensionsCore

@MainActor
final class LivePreviewCoordinator {
    private weak var adapter: HostControlMutationAdapter?
    private var transaction: PreviewTransaction<[Float]>?
    private var displayLink: CADisplayLink?
    private let coalescer = PreviewUpdateCoalescer<[Float]>()

    func begin(adapter: HostControlMutationAdapter) throws {
        self.adapter = adapter
        let transaction = PreviewTransaction(initialValue: adapter.snapshot()) { [weak adapter] values in
            _ = adapter?.apply(values)
        }
        try transaction.begin()
        self.transaction = transaction
        let link = CADisplayLink(target: self, selector: #selector(flushPending))
        link.add(to: .main, forMode: .common)
        displayLink = link
    }

    func update(_ values: [Float]) {
        Task { await coalescer.enqueue(values) }
    }

    @objc private func flushPending() {
        Task { @MainActor [weak self] in
            guard let self else { return }
            await coalescer.flush { [weak self] values in
                Task { @MainActor in try? self?.transaction?.update(values) }
            }
            self.adapter?.invalidatePreview()
        }
    }

    func commit() {
        Task { @MainActor [weak self] in
            guard let self else { return }
            await coalescer.finish { [weak self] values in
                Task { @MainActor in try? self?.transaction?.update(values) }
            }
            try? self.transaction?.commit()
            self.finish()
        }
    }

    func cancel() {
        try? transaction?.cancel()
        finish()
    }

    func selectionDidChange(to identity: ObjectIdentifier) {
        guard adapter?.selectionIdentity != identity else { return }
        try? transaction?.selectionDidChange()
        finish()
    }

    private func finish() {
        displayLink?.invalidate()
        displayLink = nil
        transaction = nil
        adapter = nil
    }
}
#endif
```

PNG_SIGNATURE = b"\x89PNG\r\n\x1a\n"

def png_dimensions(path: Path) -> tuple[int, int]:
    data = path.read_bytes()
    if len(data) < 24 or data[:8] != PNG_SIGNATURE or data[12:16] != b"IHDR":

    if width <= 0 or height <= 0:

    return width, height
