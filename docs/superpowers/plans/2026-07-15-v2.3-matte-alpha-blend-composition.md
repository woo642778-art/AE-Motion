# AE motion v2.3 Matte, Alpha, Blend and Composition Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an AE-style, non-destructive composition system with live Blend, Alpha, Channel, Track Matte, Null, Parenting and editable multi-layer Pre-compose workflows integrated contextually into the existing host editor.

**Architecture:** Add a clean-room Codable composition graph and deterministic compositing math to `AEMotionExtensionsCore`. `AEMotionExtensionsHost` maintains reversible shadow transactions for live preview and commits only through verified host UI actions or mutation callbacks. Unsupported or ambiguous host contracts fail closed and never write the host's private timeline database.

**Tech Stack:** Swift 6, Swift Package Manager, Foundation, UIKit, CoreGraphics, CoreImage where available, XCTest, Python contract tests, XcodeGen TestHost, Xcode 26.3, iOS 15+, GitHub Actions.

import Foundation

public enum CompositionValidationError: Error, Equatable {
    case missingRootComposition(UUID)
    case duplicateIdentifier(UUID)
    case invalidComposition(UUID)
    case invalidLayerTimeRange(UUID)
    case missingMatteSource(layerID: UUID, sourceID: UUID)
    case missingParent(layerID: UUID, parentID: UUID)
    case missingPrecomposition(layerID: UUID, compositionID: UUID)
}

public enum CompositionValidator {
    public static func validate(_ document: CompositionDocument) throws {
        guard document.compositions.contains(where: { $0.id == document.rootCompositionID }) else {
            throw CompositionValidationError.missingRootComposition(document.rootCompositionID)
        }

        var identifiers = Set<UUID>()
        for composition in document.compositions {
            guard identifiers.insert(composition.id).inserted else {
                throw CompositionValidationError.duplicateIdentifier(composition.id)
            }
            guard composition.duration >= 0,
                  composition.frameRate > 0,
                  composition.width > 0,
                  composition.height > 0 else {
                throw CompositionValidationError.invalidComposition(composition.id)
            }
            let layerIDs = Set(composition.layers.map(\.id))
            for layer in composition.layers {
                guard identifiers.insert(layer.id).inserted else {
                    throw CompositionValidationError.duplicateIdentifier(layer.id)
                }
                guard layer.timeRange.start.isFinite,
                      layer.timeRange.duration.isFinite,
                      layer.timeRange.duration >= 0 else {
                    throw CompositionValidationError.invalidLayerTimeRange(layer.id)
                }
                if let matte = layer.matte, !layerIDs.contains(matte.sourceLayerID) {
                    throw CompositionValidationError.missingMatteSource(layerID: layer.id, sourceID: matte.sourceLayerID)
                }
                if let parent = layer.parent, !layerIDs.contains(parent.parentLayerID) {
                    throw CompositionValidationError.missingParent(layerID: layer.id, parentID: parent.parentLayerID)
                }
                if case let .precomposition(id) = layer.kind,
                   !document.compositions.contains(where: { $0.id == id }) {
                    throw CompositionValidationError.missingPrecomposition(layerID: layer.id, compositionID: id)
                }
            }
        }
    }
}

```swift
import Foundation

public struct RGBA: Codable, Equatable, Sendable {
    public var r: Double
    public var g: Double
    public var b: Double
    public var a: Double

    public init(r: Double, g: Double, b: Double, a: Double) {
        self.r = r
        self.g = g
        self.b = b
        self.a = a
    }

    public static let clear = RGBA(r: 0, g: 0, b: 0, a: 0)

    public func clamped() -> RGBA {
        RGBA(r: r.clamped01, g: g.clamped01, b: b.clamped01, a: a.clamped01)
    }
}

private extension Double {
    var clamped01: Double { min(max(self, 0), 1) }
}

public enum BlendMathError: Error, Equatable {
    case unsupportedMode(String)
}

public enum BlendMath {
    public static func blend(source s: RGBA, destination d: RGBA, modeID: String) throws -> RGBA {
        guard BlendModeCatalogue.descriptor(id: modeID) != nil else {
            throw BlendMathError.unsupportedMode(modeID)
        }

        let rgb: (Double, Double, Double)
        switch modeID {
        case "normal": rgb = (s.r, s.g, s.b)
        case "darken": rgb = (min(s.r, d.r), min(s.g, d.g), min(s.b, d.b))
        case "multiply": rgb = (s.r * d.r, s.g * d.g, s.b * d.b)
        case "add", "linear-dodge": rgb = (s.r + d.r, s.g + d.g, s.b + d.b)
        case "lighten": rgb = (max(s.r, d.r), max(s.g, d.g), max(s.b, d.b))
        case "screen": rgb = (1 - (1 - s.r) * (1 - d.r), 1 - (1 - s.g) * (1 - d.g), 1 - (1 - s.b) * (1 - d.b))
        case "overlay": rgb = (overlay(s.r, d.r), overlay(s.g, d.g), overlay(s.b, d.b))
        case "hard-light": rgb = (overlay(d.r, s.r), overlay(d.g, s.g), overlay(d.b, s.b))
        case "difference", "classic-difference": rgb = (abs(d.r - s.r), abs(d.g - s.g), abs(d.b - s.b))
        case "exclusion": rgb = (d.r + s.r - 2 * d.r * s.r, d.g + s.g - 2 * d.g * s.g, d.b + s.b - 2 * d.b * s.b)
        case "subtract": rgb = (d.r - s.r, d.g - s.g, d.b - s.b)
        case "divide": rgb = (safeDivide(d.r, s.r), safeDivide(d.g, s.g), safeDivide(d.b, s.b))
        default:
            throw BlendMathError.unsupportedMode(modeID)
        }

        let outA = s.a + d.a * (1 - s.a)
        return RGBA(r: rgb.0, g: rgb.1, b: rgb.2, a: outA).clamped()
    }

    private static func overlay(_ source: Double, _ destination: Double) -> Double {
        destination <= 0.5 ? 2 * source * destination : 1 - 2 * (1 - source) * (1 - destination)
    }

    private static func safeDivide(_ lhs: Double, _ rhs: Double) -> Double {
        rhs == 0 ? 1 : lhs / rhs
    }
}
```

            a: component(mapping.alpha, from: value)

        let alpha = 1 - value.a
        return preserveRGB ? RGBA(r: value.r, g: value.g, b: value.b, a: alpha) : RGBA(r: alpha, g: alpha, b: alpha, a: alpha)

```swift
import Foundation

public enum CompositionGraphError: Error, Equatable {
    case missingLayer(UUID)
    case invalidSelfReference
    case dependencyCycle
}

public struct CompositionGraph: Equatable, Sendable {
    public private(set) var composition: Composition

    public init(composition: Composition) throws {
        self.composition = composition
        try validateDependencies()
    }

    public func layer(id: UUID) -> CompositionLayer? {
        composition.layers.first { $0.id == id }
    }

    public func resolvedWorldTransform(layerID: UUID) throws -> CompositionTransform {
        try resolvedWorldTransform(layerID: layerID, visiting: [])
    }

    public mutating func setParent(layerID: UUID, parentID: UUID?, preserveWorldTransform: Bool) throws {
        guard layerID != parentID else { throw CompositionGraphError.invalidSelfReference }
        guard let index = composition.layers.firstIndex(where: { $0.id == layerID }) else {
            throw CompositionGraphError.missingLayer(layerID)
        }
        if let parentID, layer(id: parentID) == nil { throw CompositionGraphError.missingLayer(parentID) }

        let world = preserveWorldTransform ? try resolvedWorldTransform(layerID: layerID) : nil
        let oldLayer = composition.layers[index]
        composition.layers[index].parent = parentID.map(ParentBinding.init(parentLayerID:))
        do {
            try validateDependencies()
            if let world {
                let parentWorld = try parentID.map { try resolvedWorldTransform(layerID: $0) }
                composition.layers[index].transform = localTransform(world: world, parentWorld: parentWorld)
            }
        } catch {
            composition.layers[index] = oldLayer
            throw error
        }
    }

    private func resolvedWorldTransform(layerID: UUID, visiting: Set<UUID>) throws -> CompositionTransform {
        guard let layer = layer(id: layerID) else { throw CompositionGraphError.missingLayer(layerID) }
        guard !visiting.contains(layerID) else { throw CompositionGraphError.dependencyCycle }
        guard let parentID = layer.parent?.parentLayerID else { return layer.transform }
        let parent = try resolvedWorldTransform(layerID: parentID, visiting: visiting.union([layerID]))
        return composed(parent: parent, child: layer.transform)
    }

    private func composed(parent: CompositionTransform, child: CompositionTransform) -> CompositionTransform {
        CompositionTransform(
            positionX: parent.positionX + child.positionX,
            positionY: parent.positionY + child.positionY,
            anchorX: child.anchorX,
            anchorY: child.anchorY,
            scaleX: parent.scaleX * child.scaleX,
            scaleY: parent.scaleY * child.scaleY,
            rotationDegrees: parent.rotationDegrees + child.rotationDegrees,
            opacity: parent.opacity * child.opacity
        )
    }

    private func localTransform(world: CompositionTransform, parentWorld: CompositionTransform?) -> CompositionTransform {
        guard let parentWorld else { return world }
        return CompositionTransform(
            positionX: world.positionX - parentWorld.positionX,
            positionY: world.positionY - parentWorld.positionY,
            anchorX: world.anchorX,
            anchorY: world.anchorY,
            scaleX: parentWorld.scaleX == 0 ? world.scaleX : world.scaleX / parentWorld.scaleX,
            scaleY: parentWorld.scaleY == 0 ? world.scaleY : world.scaleY / parentWorld.scaleY,
            rotationDegrees: world.rotationDegrees - parentWorld.rotationDegrees,
            opacity: parentWorld.opacity == 0 ? world.opacity : world.opacity / parentWorld.opacity
        )
    }

    private func validateDependencies() throws {
        for layer in composition.layers {
            _ = try resolvedWorldTransform(layerID: layer.id)
            if layer.matte?.sourceLayerID == layer.id { throw CompositionGraphError.invalidSelfReference }
        }
    }
}
```

The first implementation preserves the full scalar transform model used by the composition graph. Rotation-aware position composition must then be expanded with affine-matrix tests before host commit is enabled for nonzero parent rotation; until those tests pass, the host capability reports translation-only parenting as unavailable rather than applying incorrect compensation.

import Foundation

public enum PrecomposeError: Error, Equatable {
    case compositionNotFound(UUID)
    case requiresAtLeastTwoLayers
    case selectedLayerMissing(UUID)
    case invalidSelection
}

public struct PrecomposeRequest: Equatable, Sendable {
    public var compositionID: UUID
    public var selectedLayerIDs: [UUID]
    public var name: String

    public init(compositionID: UUID, selectedLayerIDs: [UUID], name: String) {
        self.compositionID = compositionID
        self.selectedLayerIDs = selectedLayerIDs
        self.name = name
    }
}

public struct PrecomposeResult: Equatable, Sendable {
    public var document: CompositionDocument
    public var childComposition: Composition
    public var precompLayer: CompositionLayer
    public var originalDocument: CompositionDocument
}

public enum PrecomposeEngine {
    public static func precompose(document: CompositionDocument, request: PrecomposeRequest) throws -> PrecomposeResult {
        let uniqueIDs = Array(NSOrderedSet(array: request.selectedLayerIDs)) as? [UUID] ?? request.selectedLayerIDs
        guard uniqueIDs.count >= 2 else { throw PrecomposeError.requiresAtLeastTwoLayers }
        guard let compositionIndex = document.compositions.firstIndex(where: { $0.id == request.compositionID }) else {
            throw PrecomposeError.compositionNotFound(request.compositionID)
        }

        let source = document.compositions[compositionIndex]
        let selectedSet = Set(uniqueIDs)
        let selected = source.layers.filter { selectedSet.contains($0.id) }
        guard selected.count == selectedSet.count else {
            let missing = uniqueIDs.first { id in !selected.contains(where: { $0.id == id }) }!
            throw PrecomposeError.selectedLayerMissing(missing)
        }

        let start = selected.map(\.timeRange.start).min()!
        let end = selected.map(\.timeRange.end).max()!
        guard end >= start else { throw PrecomposeError.invalidSelection }

        let moved = selected.map { layer -> CompositionLayer in
            var copy = layer
            copy.timeRange.start -= start
            return copy
        }
        let child = Composition(
            name: request.name,
            duration: end - start,
            frameRate: source.frameRate,
            width: source.width,
            height: source.height,
            layers: moved
        )
        let precomp = CompositionLayer(
            name: request.name,
            kind: .precomposition(child.id),
            timeRange: .init(start: start, duration: end - start),
            transform: .identity
        )

        let firstSelectedIndex = source.layers.firstIndex { selectedSet.contains($0.id) }!
        var parentLayers = source.layers.filter { !selectedSet.contains($0.id) }
        let insertion = min(firstSelectedIndex, parentLayers.count)
        parentLayers.insert(precomp, at: insertion)

        var updated = document
        updated.compositions[compositionIndex].layers = parentLayers
        updated.compositions.append(child)
        try CompositionValidator.validate(updated)

        return PrecomposeResult(document: updated, childComposition: child, precompLayer: precomp, originalDocument: document)
    }
}
```

import Foundation

public enum CompositionTransactionState: String, Codable, Equatable, Sendable {
    case idle
    case active
    case committed
    case cancelled
}

public enum CompositionRollbackReason: String, Codable, Equatable, Sendable {
    case userCancelled
    case selectionChanged
    case hostContractInvalidated
    case appBackgrounded
    case previewFailed
    case commitPostconditionFailed
}

public enum CompositionTransactionError: Error, Equatable {
    case alreadyStarted
    case notActive
    case invalidDocument
}

public struct CompositionCommandIntent: Codable, Equatable, Sendable {
    public var id: UUID
    public var label: String
    public var affectedCompositionID: UUID
    public var initialDocument: CompositionDocument
    public var finalDocument: CompositionDocument

    public init(id: UUID = UUID(), label: String, affectedCompositionID: UUID, initialDocument: CompositionDocument, finalDocument: CompositionDocument) {
        self.id = id
        self.label = label
        self.affectedCompositionID = affectedCompositionID
        self.initialDocument = initialDocument
        self.finalDocument = finalDocument
    }
}

public final class CompositionTransaction: @unchecked Sendable {
    public private(set) var state: CompositionTransactionState = .idle
    public private(set) var currentDocument: CompositionDocument
    public let initialDocument: CompositionDocument
    public let selectionIdentity: String
    public private(set) var rollbackReason: CompositionRollbackReason?

    public init(initialDocument: CompositionDocument, selectionIdentity: String) {
        self.initialDocument = initialDocument
        self.currentDocument = initialDocument
        self.selectionIdentity = selectionIdentity
    }

    public func begin() throws {
        guard state == .idle else { throw CompositionTransactionError.alreadyStarted }
        state = .active
    }

    public func update(_ mutation: (inout CompositionDocument) throws -> Void) throws {
        guard state == .active else { throw CompositionTransactionError.notActive }
        var candidate = currentDocument
        try mutation(&candidate)
        do { try CompositionValidator.validate(candidate) } catch { throw CompositionTransactionError.invalidDocument }
        currentDocument = candidate
    }

    public func commit(label: String, affectedCompositionID: UUID) throws -> CompositionCommandIntent {
        guard state == .active else { throw CompositionTransactionError.notActive }
        try CompositionValidator.validate(currentDocument)
        state = .committed
        return CompositionCommandIntent(label: label, affectedCompositionID: affectedCompositionID, initialDocument: initialDocument, finalDocument: currentDocument)
    }

    public func cancel(reason: CompositionRollbackReason) throws {
        guard state == .active else { throw CompositionTransactionError.notActive }
        currentDocument = initialDocument
        rollbackReason = reason
        state = .cancelled
    }

    public func selectionDidChange(to identity: String) throws {
        guard identity != selectionIdentity, state == .active else { return }
        try cancel(reason: .selectionChanged)
    }
}
```
